create schema urna;
use urna;

create table candidatos(
	idCand int not null primary key auto_increment,
    nome varchar(100),
    foto text,
    num int
);

create table votos(
	idVoto int not null primary key auto_increment,
    idCand int not null,
    foreign key (idCand) references candidatos(idCand)
);
